            default: break;
        }
        if (std::iswalnum(c)) output.push_back(c);
    }
    return output;
}

bool ParseInt64Text(const std::wstring& text, std::int64_t& out) {
    out = 0;
    if (text.empty()) return false;
    std::size_t i = 0;
    bool negative = false;
    if (text[i] == L'-') { negative = true; ++i; }
    else if (text[i] == L'+') { ++i; }
    if (i >= text.size()) return false;
    std::uint64_t value = 0;
    bool any = false;
    for (; i < text.size(); ++i) {
        const wchar_t c = text[i];
        if (c < L'0' || c > L'9') break;
        any = true;
        const std::uint64_t digit = static_cast<std::uint64_t>(c - L'0');
        if (value > (static_cast<std::uint64_t>(INT64_MAX) - digit) / 10u) return false;
        value = value * 10u + digit;
    }
    if (!any) return false;
    out = negative ? -static_cast<std::int64_t>(value) : static_cast<std::int64_t>(value);
    return true;
}

bool FloatingToInt64(double value, std::int64_t& out) {
    out = 0;
    if (!std::isfinite(value)) return false;
    const double rounded = std::round(value);
    if (std::fabs(value - rounded) > 0.001) return false;
    if (rounded < static_cast<double>(INT64_MIN) || rounded > static_cast<double>(INT64_MAX)) return false;
    out = static_cast<std::int64_t>(rounded);
    return true;
}

bool BoxedNumberToInt64(Il2CppObject* object, std::int64_t& out) {
    out = 0;
    if (!object || !g_api.object_get_class || !g_api.object_unbox || !g_api.class_get_name) return false;
    Il2CppClass* klass = g_api.object_get_class(object);
    const char* name = klass ? g_api.class_get_name(klass) : nullptr;
    void* raw = object ? g_api.object_unbox(object) : nullptr;
    if (!name || !raw) return false;
    if (Eq(name,"Double") || Eq(name,"System.Double")) return FloatingToInt64(*reinterpret_cast<double*>(raw), out);
    if (Eq(name,"Single") || Eq(name,"System.Single")) return FloatingToInt64(static_cast<double>(*reinterpret_cast<float*>(raw)), out);
    if (Eq(name,"Int64") || Eq(name,"System.Int64")) { out=*reinterpret_cast<std::int64_t*>(raw); return true; }
    if (Eq(name,"UInt64") || Eq(name,"System.UInt64")) { const auto v=*reinterpret_cast<std::uint64_t*>(raw); if(v>static_cast<std::uint64_t>(INT64_MAX)) return false; out=static_cast<std::int64_t>(v); return true; }
    if (Eq(name,"Int32") || Eq(name,"System.Int32")) { out=*reinterpret_cast<std::int32_t*>(raw); return true; }
    if (Eq(name,"UInt32") || Eq(name,"System.UInt32")) { out=*reinterpret_cast<std::uint32_t*>(raw); return true; }
    return false;
}

bool ReadFloatingField64(Il2CppObject* object, Il2CppClass* klass, const char* member, std::int64_t& out) {
    FieldInfo* field = FindField(klass, member);
    if (!field) return false;
    if (FieldType(field,"System.Double")) { double v=0; g_api.field_get_value(object,field,&v); return FloatingToInt64(v,out); }
    if (FieldType(field,"System.Single")) { float v=0; g_api.field_get_value(object,field,&v); return FloatingToInt64(static_cast<double>(v),out); }
    return false;
}

bool ReadTradeMember64(Il2CppObject* object, const char* member, std::int64_t& value) {
    value = 0;
    if (!object || !member) return false;
    Il2CppClass* klass = g_api.object_get_class(object);
    if (!klass) return false;
    if (ReadScalarMember64(object, klass, member, value)) return true;
    if (ReadFloatingField64(object, klass, member, value)) return true;
    Il2CppObject* boxed = nullptr;
    if (ReadUiMemberObject(object, klass, member, boxed) && boxed) {
        if (BoxedNumberToInt64(boxed, value)) return true;
        std::wstring boxedText;
        if (ManagedObjectText(boxed, boxedText) && ParseInt64Text(boxedText, value)) return true;
    }
    std::wstring text;
    return ReadUiMemberText(object, klass, member, text) && ParseInt64Text(text, value);
}

bool EnumerateManagedObjects(Il2CppObject* collection, std::vector<Il2CppObject*>& objects) {
    objects.clear();
    if (!collection) return false;
    if (ReadManagedPointerArray(collection, objects, 512)) return true;
    Il2CppClass* klass = g_api.object_get_class(collection);
    if (!klass) return false;
    wchar_t ignored[128]{};
    std::int32_t count = 0;
    if (ScalarGetter(klass, "get_Count", ManagedThis(collection), count, ignored, _countof(ignored)) && count >= 0 && count <= 512) {
        const MethodInfo* getItem = ExactMethod(klass, "get_Item", 1, false, "System.Int32");
        if (getItem) {
            for (std::int32_t i = 0; i < count; ++i) {
                std::int32_t index = i;
                void* args[] = {&index};
                Il2CppObject* item = nullptr;
                wchar_t itemError[96]{};
                if (!InvokeObjectArgs(getItem, ManagedThis(collection), args, item, itemError, _countof(itemError))) return false;
                if (item) objects.push_back(item);
            }
            return true;
        }
    }
    const MethodInfo* getEnumerator = FindMethod(klass, "GetEnumerator", 0);
    if (!getEnumerator) return false;
    Il2CppObject* enumerator = nullptr;
    if (!InvokeObject(getEnumerator, ManagedThis(collection), enumerator, ignored, _countof(ignored)) || !enumerator) return false;
    Il2CppClass* ec = g_api.object_get_class(enumerator);
    if (!ec) return false;
    const MethodInfo* moveNext = FindMethod(ec, "MoveNext", 0);
    const MethodInfo* getCurrent = FindMethod(ec, "get_Current", 0);
    if (!moveNext || !getCurrent) return false;
    for (int guard = 0; guard < 512; ++guard) {
        std::int64_t moved = 0;
        if (!InvokeScalar(moveNext, ManagedThis(enumerator), moved, ignored, _countof(ignored))) return false;
        if (!moved) return true;
        Il2CppObject* current = nullptr;
        if (!InvokeObject(getCurrent, ManagedThis(enumerator), current, ignored, _countof(ignored))) return false;
        if (!current) continue;
        objects.push_back(current);
        Il2CppClass* cc = g_api.object_get_class(current);
        const MethodInfo* getValue = cc ? FindMethod(cc, "get_Value", 0) : nullptr;
        if (getValue) {
            Il2CppObject* value = nullptr;
            if (InvokeObject(getValue, ManagedThis(current), value, ignored, _countof(ignored)) && value)
                objects.push_back(value);
        }
    }
    return false;
}

bool IsActiveTradeObject(Il2CppObject* object, std::int64_t& exchangeId,
                         std::int64_t& requestRoleId, std::int64_t& agreeRoleId) {
    exchangeId = 0; requestRoleId = 0; agreeRoleId = 0;
    if (!ReadTradeMember64(object, "ExchangeID", exchangeId) || exchangeId <= 0) return false;
    (void)ReadTradeMember64(object, "RequestRoleID", requestRoleId);
    (void)ReadTradeMember64(object, "AgreeRoleID", agreeRoleId);
    return true;
}

bool ResolveLuaManagerInstance(Il2CppClass* manager, Il2CppObject*& instance) {
    instance = nullptr;
    if (!manager) return false;
    for (const char* name : {"get_Instance","GetInstance","get_Current"}) {
        const MethodInfo* method = FindMethod(manager,name,0);
        if (!method || !StaticMethod(method)) continue;
        wchar_t ignored[128]{};
        if (InvokeObject(method,nullptr,instance,ignored,_countof(ignored)) && instance) return true;
    }
    return false;
}

bool TryTradeScriptObject(Il2CppObject* script, std::int64_t& exchangeId,
                          std::int64_t& requestRoleId, std::int64_t& agreeRoleId) {
    if (!script) return false;
    if (IsActiveTradeObject(script,exchangeId,requestRoleId,agreeRoleId)) return true;
    Il2CppClass* klass = g_api.object_get_class(script);
    if (!klass) return false;
    for (const char* member : {"Data","data","Table","LuaTable","Script","script","Instance","Value","Object"}) {
        Il2CppObject* child = nullptr;
        if (ReadUiMemberObject(script,klass,member,child) && child &&
            IsActiveTradeObject(child,exchangeId,requestRoleId,agreeRoleId)) return true;
    }
    return false;
}

bool ProbeViaLuaSystemManager(std::int64_t& exchangeId, std::int64_t& requestRoleId,
                              std::int64_t& agreeRoleId, bool& methodFound,
                              bool& scriptReturned, int& hasScriptValue, bool& instanceResolved) {
    exchangeId=0; requestRoleId=0; agreeRoleId=0;
    methodFound=false; scriptReturned=false; hasScriptValue=-1; instanceResolved=false;
    const Il2CppImage* image = Image();
    if (!image || !g_api.string_new) return false;
    Il2CppClass* manager = g_api.class_from_name(image,"FGStudio.LuaSystem","LuaSystemManager");
    if (!manager) return false;
    const MethodInfo* getScript = FindMethod(manager,"GetScript",1);
    methodFound = getScript != nullptr;
    if (!getScript) return false;
    Il2CppObject* instance = nullptr;
    void* target = nullptr;
    if (!StaticMethod(getScript)) {
        instanceResolved = ResolveLuaManagerInstance(manager,instance);
        if (!instanceResolved) return false;
        target = ManagedThis(instance);
    } else {
        instanceResolved = true;
    }
    Il2CppString* name = g_api.string_new("Trade");
    if (!name) return false;
    const MethodInfo* hasScript = FindMethod(manager,"HasScript",1);
    if (hasScript && StaticMethod(hasScript)==StaticMethod(getScript)) {
        void* hsArgs[] = {&name};
        std::int64_t hs = 0;
        wchar_t ignored[128]{};
        if (InvokeScalarArgs(hasScript,target,hsArgs,hs,ignored,_countof(ignored))) hasScriptValue = hs ? 1 : 0;
    }
    void* args[] = {&name};
    Il2CppObject* script = nullptr;
    wchar_t ignored[160]{};
    if (!InvokeObjectArgs(getScript,target,args,script,ignored,_countof(ignored))) return false;
    scriptReturned = script != nullptr;
    return script && TryTradeScriptObject(script,exchangeId,requestRoleId,agreeRoleId);
}

void FillTradeOpen(Response& response, std::int64_t exchangeId,
                   std::int64_t requestRoleId, std::int64_t agreeRoleId) {
    response.resultCode |= kTradeProbeUiExists | kTradeProbeActiveKnown |
                           kTradeProbeActive | kTradeProbeExchangeIdKnown;
    response.value0 = 1;
    response.value1 = 1;
    response.value64_0 = exchangeId;
    response.value64_1 = requestRoleId ? requestRoleId : agreeRoleId;
}

// Trade V3: first inspect LuaSystemManager.GetScript("Trade") because the shipped
// runtime documentation explicitly exposes GetScript/HasScript as script-lifetime
// accessors. Also accept Lua numbers boxed as Double/Single. UI enumeration remains
// a secondary observer only.
bool ProbeTradeState(Response& response, wchar_t* detail, std::size_t cap) {
    if (!EnsureUiLua(true, detail, cap)) return false;
    response = {};
    response.resultCode = kTradeProbeFindUiOk;

    std::int64_t exchangeId=0, requestRoleId=0, agreeRoleId=0;
    bool managerMethod=false, scriptReturned=false, managerInstance=false;
    int hasScript=-1;
    if (ProbeViaLuaSystemManager(exchangeId,requestRoleId,agreeRoleId,
                                 managerMethod,scriptReturned,hasScript,managerInstance)) {
        FillTradeOpen(response,exchangeId,requestRoleId,agreeRoleId);
        SetText(detail,cap,L"V3 GetScript('Trade') ACTIVE • ExchangeID=");
        AppendInt64(detail,cap,exchangeId);
        Append(detail,cap,L" • RequestRoleID="); AppendInt64(detail,cap,requestRoleId);
        Append(detail,cap,L" • AgreeRoleID="); AppendInt64(detail,cap,agreeRoleId);
        return true;
    }

    std::vector<Il2CppObject*> objects;
    bool findAllOk=false;
    const MethodInfo* findAll = ExactMethod(g_ui.guiApi,"FindAllUIs",0,true);
    const wchar_t* allSource=L"FindAllUIs";
    if (!findAll) { findAll=ExactMethod(g_ui.guiApi,"MainFindAllUIs",0,true); allSource=L"MainFindAllUIs"; }
    if (findAll) {
        Il2CppObject* all=nullptr; wchar_t ignored[128]{};
        if (InvokeObject(findAll,nullptr,all,ignored,_countof(ignored)) && all)
            findAllOk=EnumerateManagedObjects(all,objects);
    }
    for (Il2CppObject* object : objects) {
        if (!IsActiveTradeObject(object,exchangeId,requestRoleId,agreeRoleId)) continue;
        FillTradeOpen(response,exchangeId,requestRoleId,agreeRoleId);
        SetText(detail,cap,L"V3 "); Append(detail,cap,allSource); Append(detail,cap,L" ACTIVE • ExchangeID=");
        AppendInt64(detail,cap,exchangeId);
        return true;
    }

    bool namedTradeObject=false;
    const MethodInfo* findUi=ExactMethod(g_ui.guiApi,"FindUI",1,true,"System.String");
    if (!findUi) findUi=ExactMethod(g_ui.guiApi,"MainFindUI",1,true,"System.String");
    if (findUi) {
        Il2CppString* tradeName=g_api.string_new("Trade");
        if (tradeName) {
            Il2CppObject* trade=nullptr; void* args[]={&tradeName}; wchar_t ignored[128]{};
            if (InvokeObjectArgs(findUi,nullptr,args,trade,ignored,_countof(ignored)) && trade) {
                namedTradeObject=true;
                if (TryTradeScriptObject(trade,exchangeId,requestRoleId,agreeRoleId)) {
                    FillTradeOpen(response,exchangeId,requestRoleId,agreeRoleId);
                    SetText(detail,cap,L"V3 FindUI('Trade') ACTIVE • ExchangeID="); AppendInt64(detail,cap,exchangeId);
                    return true;
                }
            }
        }
    }

    SetText(detail,cap,L"V3 idle • GetScriptMethod="); Append(detail,cap,managerMethod?L"YES":L"NO");
    Append(detail,cap,L" Instance="); Append(detail,cap,managerInstance?L"OK":L"NO");
    Append(detail,cap,L" HasScript=");
    if (hasScript<0) Append(detail,cap,L"?"); else Append(detail,cap,hasScript?L"1":L"0");
    Append(detail,cap,L" ScriptObj="); Append(detail,cap,scriptReturned?L"YES":L"NO");
    Append(detail,cap,L" • "); Append(detail,cap,allSource);
    Append(detail,cap,findAllOk?L" OK objects=":L" FAIL objects="); AppendInt(detail,cap,static_cast<int>(objects.size()));
    Append(detail,cap,namedTradeObject?L" • FindUI object":L" • FindUI null");
    return true;
}

bool EnsureShared() {
    if (g_shared) return true;
    wchar_t name[96]{}; MappingName(GetCurrentProcessId(), name, _countof(name));
    g_mapping = OpenFileMappingW(FILE_MAP_ALL_ACCESS, FALSE, name);
    if (!g_mapping) return false;
    g_shared = reinterpret_cast<SharedBlock*>(MapViewOfFile(g_mapping, FILE_MAP_ALL_ACCESS, 0, 0, sizeof(SharedBlock)));
    if (!g_shared || g_shared->magic != kMagic || g_shared->protocolVersion != kProtocolVersion ||
        g_shared->targetPid != GetCurrentProcessId()) {
        if (g_shared) UnmapViewOfFile(g_shared);
        if (g_mapping) CloseHandle(g_mapping);
        g_shared=nullptr; g_mapping=nullptr; return false;
    }
    InterlockedExchange(&g_shared->bridgeLoaded,1);
    return true;
}

void ProcessRequest() {
    if (!EnsureShared()) return;
    const LONG seq=g_shared->requestSeq;
    if (seq<=0 || seq==g_shared->completedSeq) return;
    if (InterlockedCompareExchange(&g_shared->bridgeBusy,1,0)!=0) return;
    Response r{}; wchar_t detail[512]{}; bool ok=false;
    if (GetCurrentThreadId()!=g_shared->targetWindowThreadId) {
        SetText(detail,_countof(detail),L"Sai callback thread; action bị chặn");
    } else {
        const Command cmd=static_cast<Command>(g_shared->request.command);
        switch(cmd) {
            case Command::ReadBagPage: ok=ReadBagPage(g_shared->request.arg0,r,detail,_countof(detail)); break;
            case Command::ClickInternalPointRawTest: ok=ClickInternalPointRawTest(g_shared->request.arg0,g_shared->request.arg1,r,detail,_countof(detail)); break;
            case Command::DragInternalPoint: ok=DragInternalPoint(g_shared->request.arg0,g_shared->request.arg1,g_shared->request.arg2,r,detail,_countof(detail)); break;
            case Command::ClickItemSellAction: ok=ClickItemSellAction(r,detail,_countof(detail)); break;
            case Command::ProbeItemSellAction: ok=ProbeItemSellAction(r,detail,_countof(detail)); break;
            case Command::ProbeTradeState: ok=ProbeTradeState(r,detail,_countof(detail)); break;
            default: SetText(detail,_countof(detail),L"Command test không hợp lệ"); break;
        }
    }
    r.ok=ok?1:0; SetText(r.detail,_countof(r.detail),detail); g_shared->response=r;
    MemoryBarrier(); InterlockedExchange(&g_shared->completedSeq,seq); InterlockedExchange(&g_shared->bridgeBusy,0);
}

extern "C" __declspec(dllexport) LRESULT CALLBACK TlcGetMessageHook(int code, WPARAM wParam, LPARAM lParam) {
    (void)wParam;
    if (code>=0 && lParam) {
        const MSG* msg=reinterpret_cast<const MSG*>(lParam);
        if (msg->message==kWakeMessage) ProcessRequest();
    }
    return CallNextHookEx(nullptr,code,wParam,lParam);
}
