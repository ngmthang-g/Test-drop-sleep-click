int ScaleCoord(int value,int fromExtent,int toExtent) {
    if(fromExtent<=0||toExtent<=0) return value;
    return static_cast<int>((static_cast<long long>(value)*toExtent+fromExtent/2)/fromExtent);
}

bool ResolvePoint(const ClickStep& p,HWND hwnd,int& x,int& y) {
    if(!p.valid) return false;
    RECT rc{}; if(!GetClientRect(hwnd,&rc)) return false;
    const int w=rc.right-rc.left,h=rc.bottom-rc.top; if(w<=0||h<=0) return false;
    x=p.x; y=p.y;
    if(p.baseW>0&&p.baseH>0&&(p.baseW!=w||p.baseH!=h)) { x=ScaleCoord(x,p.baseW,w); y=ScaleCoord(y,p.baseH,h); }
    x=std::clamp(x,0,w-1); y=std::clamp(y,0,h-1); return true;
}

int NormalizeCoord(int v,int extent) {
    if(extent<=0) return -1;
    return static_cast<int>((static_cast<long long>(std::clamp(v,0,extent-1))*kCoordinateScale)/extent);
}

bool ParseStep(const std::string& text,ClickStep& p) {
    std::stringstream ss(text); std::string item; std::array<int,6> v{}; int i=0;
    while(std::getline(ss,item,',')&&i<6){try{v[static_cast<std::size_t>(i++)]=std::stoi(item);}catch(...){return false;}}
    if(i!=6)return false;
    p={v[0],v[1],v[2],v[3],v[5]!=0}; return true;
}

bool LoadTlScan(const std::wstring& path,ScanConfig& c,std::wstring& detail) {
    std::ifstream in(std::filesystem::path(path),std::ios::binary); if(!in){detail=L"không thấy "+path;return false;}
    std::string line; if(!std::getline(in,line)||(line!="TL_SCAN_V5"&&line!="TL_SCAN_AUTO_V1"&&line!="TL_SCAN_AUTO_V2")){detail=L"file .tlscan không đúng TL_SCAN_V5/TL_SCAN_AUTO";return false;}
    int loaded=0;
    while(std::getline(in,line)){
        if(!line.empty()&&line.back()=='\r')line.pop_back(); const auto eq=line.find('='); if(eq==std::string::npos)continue;
        const std::string key=line.substr(0,eq),val=line.substr(eq+1);
        if(key=="swipe_start")ParseStep(val,c.swipeStart);
        else if(key=="swipe_end")ParseStep(val,c.swipeEnd);
        else if(key.rfind("step",0)==0&&key!="step_count"){
            try{const int idx=std::stoi(key.substr(4)); if(idx>=0&&idx<42&&ParseStep(val,c.steps[static_cast<std::size_t>(idx)]))++loaded;}catch(...){}
        }
    }
    if(loaded<42){detail=L".tlscan chỉ có "+std::to_wstring(loaded)+L"/42 tọa hợp lệ";return false;}
    if(!c.swipeStart.valid||!c.swipeEnd.valid){detail=L".tlscan thiếu swipe_start/swipe_end";return false;}
    detail=L"đã nạp nguyên 42 tọa + swipe từ TL_SCAN_V5"; return true;
}

bool HasModule(DWORD pid,const wchar_t* name) {
    HANDLE snap=CreateToolhelp32Snapshot(TH32CS_SNAPMODULE|TH32CS_SNAPMODULE32,pid); if(snap==INVALID_HANDLE_VALUE)return false;
    MODULEENTRY32W e{};e.dwSize=sizeof(e);bool found=false;
    if(Module32FirstW(snap,&e)){do{if(_wcsicmp(e.szModule,name)==0){found=true;break;}}while(Module32NextW(snap,&e));}
    CloseHandle(snap); return found;
}
BOOL CALLBACK EnumGameWindows(HWND hwnd,LPARAM param) {
    if(!IsWindowVisible(hwnd)||GetWindowTextLengthW(hwnd)<=0)return TRUE;
    DWORD pid=0;DWORD tid=GetWindowThreadProcessId(hwnd,&pid);if(!pid||!tid||!HasModule(pid,kGameModule))return TRUE;
    auto* out=reinterpret_cast<std::vector<GameClient>*>(param);for(const auto& g:*out)if(g.pid==pid)return TRUE;
    wchar_t title[512]{};GetWindowTextW(hwnd,title,_countof(title));out->push_back({pid,tid,hwnd,title});return TRUE;
}
